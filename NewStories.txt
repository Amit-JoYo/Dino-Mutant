צַלָּם בָּא לַבַּיִת שֶׁל תָּמָר

הַיּוֹם מִצְטַלְּמִים!
אַבָּא, אִמָּא, בָּר וְתָמָר עוֹמְדִים
לְיַד הַחַלּוֹן.
פִּתְאוֹם תָּמָר זָזָה יָמִינָה.
הַצַּלָּם אָמַר: "תָּמָר, בְּבַקָּשָׁה לַעֲמֹד!"
מִינִי נָבְחָה וְקָפְצָה עַל הַסַּפָּה.
הַצַּלָּם אָמַר: "מִינִי, בְּבַקָּשָׁה לֹא לִנְבֹּחַ וְלֹא לִקְפֹּץ!"
פִּתְאוֹם סָבְתָא בָּאָה.
תָּמָר שָׂמְחָה: "סָבְתָא! תַּעַמְדִי לְיָדִי!"
הַצַּלָּם אָמַר: "עַכְשָׁו מִצְטַלְּמִים. אַחַת, שְׁתַּיִם... קְלִיק!
צִלַּמְתִּי אֶת הַמִּשְׁפָּחָה הֲכִי חֲבִיבָה בָּעוֹלָם!"
------------------------------------------------

הַשִּׁטְפוֹנוֹת בַּמִּדְבָּר

בַּמִּדְבָּר יוֹרְדִים גְּשָׁמִים
רַק לְעִתִּים רְחוֹקוֹת,
וְלֹא זוֹרְמִים מַיִם בַּנְּחָלִים
רֹב יְמוֹת הַשָּׁנָה.
אֲנָשִׁים מְטַיְּלִים בַּנְּחָלִים וְרוֹאִים אֶת
הַצְּמָחִים וְאֶת הַחַיּוֹת.
אֲבָל לִפְעָמִים יוֹרְדִים גְּשָׁמִים חֲזָקִים
בַּמְּקוֹמוֹת הַגְּבוֹהִים בַּמִּדְבָּר, וְאָז הֲמוֹן מַיִם
זוֹרְמִים לַנְּחָלִים. הַנְּחָלִים מִתְמַלְּאִים
בְּבַת אַחַת בְּמַיִם וְיֵשׁ שִׁטָּפוֹן.
בִּזְמַן הַשִּׁטָּפוֹן יֵשׁ סַכָּנָה לַמְּטַיְּלִים בַּנַּחַל.
הַמְּטַיְּלִים צְרִיכִים לַעֲלוֹת מִיָּד לְמָקוֹם
גָּבוֹהַּ וְלִקְרֹא לַמִּשְׁטָרָה. אִם הַסַּכָּנָה גְּדוֹלָה,
הַמִּשְׁטָרָה מַזְעִיקָה מַסּוֹק, וְהָאֲנָשִׁים בַּמַּסּוֹק
מְחַלְּצִים אֶת הַמְּטַיְּלִים.
---------------------------------------------------

שֶׁלֶג בַּגָּלִיל
   
בַּלַּיְלָה הָיָה קַר וְיָרַד שֶׁלֶג בַּגָּלִיל.
אִמָּא אָמְרָה: "מָחָר נִסַּע לַגָּלִיל לִרְאוֹת אֶת הַשֶּׁלֶג."
אֲבָל כַּאֲשֶׁר הַמִּשְׁפָּחָה הִתְעוֹרְרָה,
הַשֶּׁמֶשׁ זָרְחָה וְהָאֲוִיר הִתְחַמֵּם.
אַבָּא הִדְלִיק אֶת הָרַדְיוֹ וְשָׁמַע:
"הַשֶּׁלֶג בַּגָּלִיל כְּבָר הִפְשִׁיר!"
------------------------------------------------------

לְבַד בַּגֶּשֶׁם
   
שַׁחַר עָמַד לְיַד הַחַלּוֹן וְהִבִּיט אֶל הָרְחוֹב.
הַגֶּשֶׁם יָרַד בְּלִי הַפְסָקָה.
"אַבָּא, מְשַׁעֲמֵם לִי," אָמַר שַׁחַר.
אַבָּא עָנָה: "אַתָּה יָכוֹל לְשַׂחֵק עִם מַיָה!"
אֲבָל שַׁחַר לֹא רָצָה לְשַׂחֵק עִם מַיָה.
שַׁחַר הִמְשִׁיךְ לַעֲמֹד לְיַד הַחַלּוֹן.
פִּתְאוֹם שַׁחַר רָאָה כֶּלֶב קָטָן עוֹמֵד לְיַד הַשַּׁעַר.
שַׁחַר קָרָא: "אַבָּא, תִּרְאֶה, יֵשׁ כֶּלֶב קָטָן לְיַד הַשַּׁעַר!"
אַבָּא הִתְקָרֵב לַחַלּוֹן וְאָמַר: "כְּלַבְלַב מִסְכֵּן,
לְבַדּוֹ בַּגֶּשֶׁם... צָרִיךְ לְהַכְנִיס אוֹתוֹ הַבַּיְתָה!"
אַבָּא לָקַח אֶת הַכֶּלֶב הַקָּטָן, נִגֵּב אוֹתוֹ וְהִנִּיחַ
אוֹתוֹ עַל שְׂמִיכָה יְשָׁנָה. אַחַר כָּךְ שַׁחַר נָתַן לַכֶּלֶב חָלָב חַם.
הַכֶּלֶב שָׂמַח וְגַם שַׁחַר שָׂמַח.
-----------------------------------------------------------

אֵיךְ מְגַדְּלִים יְרָקוֹת?

עוֹשִׂים חוֹרִים קְטַנִּים בָּאֲדָמָה.
שָׂמִים בְּתוֹךְ הַחוֹרִים זְרָעִים שֶׁל יְרָקוֹת.
מְכַסִּים אֶת הַזְּרָעִים בָּאֲדָמָה.
מַשְׁקִים אֶת הָאֲדָמָה. מְחַכִּים.
כַּעֲבֹר כַּמָּה יָמִים מְצִיצִים
עָלִים קְטַנְטַנִּים מִן הָאֲדָמָה.
אֵלֶּה הַנְּבָטִים.
הַנְּבָטִים צוֹמְחִים וְהַיְּרָקוֹת גְּדֵלִים
עַד שֶׁאֶפְשָׁר לִקְטֹף אוֹתָם.
לִפְעָמִים אוֹכְלִים אֶת הַשֹּׁרֶשׁ,
כְּמוֹ צְנוֹנִית וְגֶזֶר.
לִפְעָמִים אוֹכְלִים אֶת הַפְּרִי,
כְּמוֹ מְלַפְפוֹן וְעַגְבָנִיָּה.
וְלִפְעָמִים אוֹכְלִים עָלִים,
כְּמוֹ חַסָּה וָתֶרֶד.
------------------------------------------------------------

גִּנָּה בַּכִּתָּה (א)

בֹּקֶר אֶחָד הַמּוֹרָה אוֹרִית
חִלְּקָה לַיְּלָדִים שַׂקִּיּוֹת קְטַנּוֹת.
אוֹרִית אָמְרָה: "בַּשַּׂקִּיּוֹת יֵשׁ זְרָעִים.
נְגַדֵּל יְרָקוֹת בַּכִּתָּה כְּמוֹ בַּגִּנָּה!"
בָּרָק שָׂמַח: "אֵיזֶה יֹפִי! נָכִין סָלָט יְרָקוֹת בַּכִּתָּה!"
אַיָּלָה זָרְעָה פִּלְפֵּל. בָּרָק זָרַע בָּצָל יָרֹק.   
מוֹר זָרְעָה חַסָּה, וְאוֹר זָרְעָה עַגְבָנִיָּה.

------------------------------------------------------------

גִּנָּה בַּכִּתָּה (ב)

עָבַר יוֹם וְעוֹד יוֹם, אֲבָל דָּבָר לֹא קָרָה.
לַיְלָה אֶחָד מוֹר חָלְמָה שֶׁפֵיָה קְטַנָּה בִּקְּרָה
בַּכִּתָּה. הַפֵיָה הִסְתַּכְּלָה סָבִיב סָבִיב וְאָז
הִתְקָרְבָה לָאֲדָנִית שֶׁל מוֹר. הַפֵיָה פִּזְּרָה
אַבְקַת קְסָמִים בָּאֲדָנִית שֶׁלָּהּ וְנֶעֶלְמָה.
בַּבֹּקֶר מוֹר מִהֲרָה לַכִּתָּה...
מָה רוֹאוֹת עֵינֶיהָ? נְבָטִים בָּאֲדָנִית שֶׁלָּהּ!
מוֹר שָׂמְחָה מְאוֹד. אֲבָל כַּאֲשֶׁר הִסְתַּכְּלָה סָבִיב,
רָאֲתָה נְבָטִים גַּם בָּאֲדָנִיּוֹת שֶׁל שְׁאָר הַיְּלָדִים.

------------------------------------------------------------

אוֹרֵחַ בַּכִּתָּה (א)

שְׁלוֹמִי נִכְנַס לַכִּיתָה. הָיָה לוֹ בַּיָּד אַרְגָּז קָטָן.
"אֲנִי רוֹצֶה לְסַפֵּר לָכֶם מָה יֵשׁ לִי בָּאַרְגָּז,"
אָמַר שְׁלוֹמִי.
"הַיּוֹם בְּחָמֵשׁ בַּבֹּקֶר הָיָה עֲדַיִן חֹשֶׁךְ.
אַבָּא שֶׁלִּי נִכְנַס לַמְּכוֹנִית וְהִדְלִיק אֶת הַפָּנָסִים.
פִּתְאוֹם אַבָּא רָאָה חַיָּה קְטַנָּה. בְּאֶמְצַע הַכְּבִישׁ!
אַבָּא עָצַר וְיָצָא מֵהַמְּכוֹנִית. הַחַיָּה הַקְּטַנָּה
הָלְכָה לְאַט לְאַט. אַבָּא הִתְקָרֵב.
זֶה הָיָה קִפּוֹד קָטָן! לַקִּפּוֹד הָיָה פֶּצַע בָּרֶגֶל.
יָרַד לוֹ דָּם.   
אַבָּא הוֹצִיא מֵהַמְּכוֹנִית אַרְגָּז, הֵרִים אֶת הַקִּפּוֹד
בִּשְׁתֵּי יָדַיִם וְשָׂם אוֹתוֹ בָּאַרְגָּז.
אַבָּא אָמַר: 'קִפּוֹד קָטָן, אַל תִּפְחַד, אַתָּה תַּבְרִיא!'"
הַמּוֹרָה אוֹרִית אָמְרָה: "בַּהַפְסָקָה נִתְקַשֵּׁר
לַסַּפָארִי וְנִשְׁאַל אֵיךְ לַעֲזֹר לַקִּפּוֹד."
------------------------------------------------------------

אוֹרֵחַ בַּכִּתָּה (ב)

הַמּוֹרָה אוֹרִית הִתְקַשְּׁרָה לַסַּפָארִי.
הִיא שָׁאֲלָה אֶת הַוֵּטֵרִינָר מָה עוֹשִׂים
כְּשֶׁמּוֹצְאִים חַיָּה שֶׁנִּפְצְעָה. הַוֵּטֵרִינָר שָׁלַח
לַמּוֹרָה בַּטֶּלֶפוֹן דַּף הוֹרָאוֹת.
מִתְקַשְּׁרִים לַמּוֹקֵד 3639 כְּדֵי לְדַוֵּחַ עַל חַיָּה שֶׁנִּפְצְעָה.*   
מְרַפְּדִים אַרְגַּז קַרְטוֹן בְּמַגֶּבֶת אוֹ בְּעָלִים.
מְחוֹרְרִים אֶת הָאַרְגָּז כְּדֵי שֶׁיִּהְיֶה לַחַיָּה אֲוִיר.
מַכְנִיסִים אֶת הַחַיָּה לָאַרְגָּז וְסוֹגְרִים כְּדֵי לִשְׁמֹר עָלֶיהָ.
לוֹקְחִים אֶת הַחַיָּה לַסַּפָארִי אוֹ לְבֵית חוֹלִים לְחַיּוֹת.
כל המילים והניקוד בעמוד זה ברורים לחלוטין וללא חיתוכים.

------------------------------------------------------------

צַלֶּמֶת טֶבַע

בְּיוֹם הַמִּשְׁפָּחָה בָּאָה לַכִּתָּה וֶרֶד, אִמָּא שֶׁל
אַיָּלָה, וְסִפְּרָה עַל הַמִּקְצוֹעַ שֶׁלָּהּ.
וֶרֶד הִיא צַלֶּמֶת טֶבַע.
הִיא אוֹהֶבֶת לְצַלֵּם בַּעֲלֵי חַיִּים, צְמָחִים וְנוֹפִים.
בַּחֹרֶף שֶׁעָבַר הִיא נָסְעָה לְאַפְרִיקָה וְצִלְּמָה
שָׁם פִּילִים, אֲרָיוֹת וְזֶבְּרוֹת.
הַשָּׁנָה, אִמָּא שֶׁל אַיָּלָה מְצַלֶּמֶת בְּיִשְׂרָאֵל.
בַּסְּתָו הִיא נָסְעָה לַנֶּגֶב וְצִלְּמָה שָׁם יְעֵלִים
מְדַלְּגִים בֵּין סְלָעִים.
וֶרֶד מְצַלֶּמֶת אֶת הַחַיּוֹת בַּמַּצְלֵמָה הַגְּדוֹלָה
שֶׁלָּהּ וְשׁוֹלַחַת סִרְטוֹנִים לְאַיָּלָה.
פַּעַם הִיא שָׁלְחָה לְאַיָּלָה סִרְטוֹן שֶׁרוֹאִים בּוֹ
עֹפֶר יוֹנֵק מֵאִמָּא שֶׁלּוֹ, הָאַיָּלָה. אַיָּלָה הִתְרַגְּשָׁה
מְאוֹד וְשָׁלְחָה לְאִמָּא סִרְטוֹן שֶׁל הָאַיָּלָה שֶׁלָּהּ.

------------------------------------------------------------

בַּקּוֹנְצֶרְט

תָּמָר הָלְכָה עִם סַבָּא שֶׁלָּהּ
לַקּוֹנְצֶרְט. תָּמָר הִתְרַגְּשָׁה.
הִיא אַף פַּעַם לֹא הָיְתָה
בְּקוֹנְצֶרְט.
אַחֲרֵי הַקּוֹנְצֶרְט הִיא סִפְּרָה
לְמוֹר עַל כָּל מָה שֶׁרָאֲתָה:
"לִפְנֵי שֶׁמַּתְחִיל הַקּוֹנְצֶרְט,
נַגָּנֵי הַתִּזְמֹרֶת עוֹלִים לַבָּמָה. הֵם מִתְכּוֹנְנִים
לִנְגִינָה וְהַקָּהָל מְחַכֶּה בְּשֶׁקֶט. הַמְּנַצֵּחַ נִכְנָס,
וְאָז הַקָּהָל מוֹחֵא כַּפַּיִם. הַמְּנַצֵּחַ מֵרִים אֶת
הַשַּׁרְבִיט, וְהַנַּגָּנִים מַתְחִילִים לְנַגֵּן.
לִפְעָמִים רַק כְּלִי אֶחָד מְנַגֵּן אֶת הַמַּנְגִּינָה,
וְלִפְעָמִים הַרְבֵּה כֵּלִים מְנַגְּנִים יַחַד.
בְּסוֹף הַקּוֹנְצֶרְט הַמְּנַצֵּחַ וְהַנַּגָּנִים
מִשְׁתַּחֲוִים לַקָּהָל.
הַמְּנַצֵּחַ יוֹרֵד רִאשׁוֹן מֵהַבָּמָה,
וְאַחֲרָיו יוֹרְדִים הַנַּגָּנִים."
