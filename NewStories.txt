טְרֶמְפּ עַל מַטְאֲטֵא - ג'וּלְיָה דּוֹנַלְדְסוֹן - אַקְסֶל שֶׁפְלֶר

מְכַשֵּׁפָה עַל הַגֹּבַהּ, עִם חָתוּל וְעִם כּוֹבַע
עִם אַף לְתִפְאֶרֶת וְצַמָּה קְשׁוּרָה בְּסֶרֶט,
הֶחָתוּל נָהַם בְּנַחַת, הִיא חִיְּכָה בְּכָל פֶּה
כְּשֶׁטָּסוּ עִם הָרוּחַ, עַל מַקֵּל שֶׁל מַטְאֲטֵא.

עַד שֶׁלְּפֶתַע פִּתְאוֹם, הָרוּחַ הִתְחַלְּפָה
"אוּפְּס..." הַכּוֹבַע עָף לוֹ, מֵרֹאשׁ הַמְּכַשֵּׁפָה.
"הִכּוֹנוּ לִנְחִיתָה!" קָרְאָה הַמְּכַשֵּׁפָה,
וְהֵם חִפְּשׂוּ, וְחִפְּשׂוּ, וְחִפְּשׂוּ, לֹא מָצְאוּ, שׁוּם כּוֹבַע בַּסְּבִיבָה.

וְאָז מִן הַשִּׂיחִים, נִשְׁמַע קִשְׁקוּשׁ זָנָב,
וְכֶלֶב מְקַפֵּץ לוֹ, עִם כּוֹבַע בֵּין שִׁנָּיו.
בְּרֹב נִימוּס הוֹשִׁיט לָהּ וְאָמַר: "בְּבַקָּשָׁה".
(בְּעוֹד הִיא מְהַדֶּקֶת אֶת הַכּוֹבַע לְרֹאשָׁהּ)
"אַגַּב", הוֹסִיף בְּקוֹל מְתַקְתַּק וּמְפַתֶּה,
"אוּלַי אֶפְשָׁר לִתְפֹּס אֶצְלֵךְ טְרֶמְפּ עַל מַטְאֲטֵא?"
"כֵּן!" אָמְרָה הַמְּכַשֵּׁפָה וְהַכְּלַבְלַב קִפֵּץ,
וְ... עוּף! שְׁלָשְׁתָּם הִמְרִיאוּ עַל מַטְאֲטֵא הָעֵץ.

מֵעַל שָׂדוֹת, מֵעַל חָרְשׁוֹת הֵם טָסוּ וְחָלְפוּ,
קִשְׁקֵשׁ הַכֶּלֶב בַּזָּנָב, וְהָרוּחוֹת נָשְׁבוּ,
הַמְּכַשֵּׁפָה אָחֲזָה הֵיטֵב בַּכּוֹבַע הַמְּחֻדָּד,
אַךְ אוֹי לֹא, הַסֶּרֶט עָף לוֹ וְאָבַד!
"הִכּוֹנוּ לִנְחִיתָה!" קָרְאָה הַמְּכַשֵּׁפָה.
וְהֵם חִפְּשׂוּ, חִפְּשׂוּ, וְלֹא מָצְאוּ שׁוּם סֶרֶט בַּסְּבִיבָה.

לְפֶתַע מִן הָעֵץ נִשְׁמַע מַשַּׁק כַּנְפֵי צִפּוֹר,
עוֹרֵב יָרֹק הוֹפִיעַ עִם סֶרֶט בַּמָּקוֹר.
בְּרֹב נִימוּס, הֵשִׁיב לָהּ אֶת הַסֶּרֶט וְאָמַר,
(בְּעוֹד הִיא מְסַדֶּרֶת אֶת הַסֶּרֶט בַּשֵּׂעָר)
בְּקוֹל הֲכִי חָבִיב הוּא מְצַיֵּץ וּמְפַתֶּה:
"אוּלַי אוּכַל לִתְפֹּס אֶצְלֵךְ טְרֶמְפּ עַל מַטְאֲטֵא?"
"וַדַּאי!" קָרְאָה הַמְּכַשֵּׁפָה, וְהָעוֹרֵב קִפֵּץ,
וְכָךְ אַרְבַּעְתָּם רָכְבוּ עַל מַטְאֲטֵא הָעֵץ.

מֵעַל בִּצּוֹת וּנְחָלִים, הֵם טָסוּ וְחָלְפוּ,
צָהַל לוֹ הָעוֹרֵב: "קְרָע! קְרָע!" וְהָרוּחוֹת נָשְׁבוּ.
הַמְּכַשֵּׁפָה תָּפְסָה הֵיטֵב בַּסֶּרֶט הַמְּנֻקָּד,
אֲבָל עַכְשָׁו שַׁרְבִיט הַקֶּסֶם עָף לוֹ וְאָבַד!
"הִכּוֹנוּ לִנְחִיתָה!" פָּקְדָה הַמְּכַשֵּׁפָה,
אַךְ שׁוּם שַׁרְבִיט שֶׁל קֶסֶם לֹא נִמְצָא כָּאן בַּסְּבִיבָה.

לְפֶתַע מִן הַמַּיִם תִּרְאוּ מִי מִזְדַּקֵּף?
צְפַרְדֵּעַ וּבְיָדָהּ שַׁרְבִיט רָטֹב וּמְטַפְטֵף.
בְּרֹב נִימוּס הֶחְזִירָה אֶת הָאֲבֵדָה,
הַמְּכַשֵּׁפָה יִבְּשָׁה אֶת הַשַּׁרְבִיט בִּקְצֵה בִּגְדָהּ.
עַכְשָׁו נִשְׁמַע קִרְקוּר גַּנְדְּרָנִי וּמְפַתֶּה:
"אוּלַי אֶפְשָׁר לִתְפֹּס אֶצְלֵךְ טְרֶמְפּ עַל מַטְאֲטֵא?"
"אֶפְשָׁר!" שָׁמְעָה צְפַרְדֵּעַ וְהֵחֵלָּה לְטַפֵּס,
וַחֲמִשְׁתָּם הִמְרִיאוּ עַל מַטְאֲטֵא הָעֵץ.

וְכָךְ הִמְשִׁיכוּ הָלְאָה בֵּין הַר וּבֵין מִדְבָּר,
קָפְצָה צְפַרְדֵּעַ מִשִּׂמְחָה אַךְ... הַמַּטְאֲטֵא נִשְׁבַּר!
חָתוּל, כְּלַבְלַב, צְפַרְדֵּעַ בְּאֶמְצַע הַטִּיסָה,
צָנְחוּ נָפְלוּ הִתְגַּלְגְּלוּ, יָשָׁר לְתוֹךְ בִּצָּה.
וְעַל חֲצִי הַמַּטְאֲטֵא נוֹתְרָה הַמְּכַשֵּׁפָה,
פִּתְאוֹם קוֹל שְׁאָגָה נוֹרָא נָהַם רָעַם סְבִיבָהּ:
"אֲנִי דְּרָקוֹן בֶּן רֶשַׁע!" פָּתַח הוּא בִּצְרָחָה,
"וּמִתְכּוֹנֵן לִזְלֹל מְכַשֵּׁפָה לַאֲרוּחָה!"
"לֹאאאא!" קָרְאָה הַמְּכַשֵּׁפָה.

בּוֹרַחַת עַל רִכְבָּהּ וְאַחֲרֶיהָ דְּרָקוֹן יוֹרֵק אֵשׁ לֶהָבָה,
"הַצִּילוּ!" צוֹרַחַת, מִיָּד הִיא מַנְחִיתָה.
וְהִיא חִפְּשָׂה חִפְּשָׂה וְאַף אֶחָד לֹא בָּא לְעֶזְרָתָהּ.
בֵּינְתַיִם, הַדְּרָקוֹן כְּבָר כָּאן, שִׁנָּיו כְּבָר חֲשׂוּפוֹת:
"מְמְמ... הַיּוֹם אֶזְלֹל מְכַשֵּׁפָה עִם כָּל הַתּוֹסָפוֹת!"

אַךְ בְּדִיּוּק שֶׁהַדְּרָקוֹן פָּתַח בַּסְּעוּדָה,
מִן הַבִּצָּה הִתְרוֹמְמָה מִפְלֶצֶת מַחֲרִידָה.
גְּבוֹהָה, כֵּהָה, וּמְלֵאָה קוֹצִים, נוֹצוֹת, כְּנָפַיִם,
עִם אַרְבָּעָה רָאשִׁים וַהֲמוֹן זוּגוֹת טְלָפַיִם.
כְּשֶׁהִשְׁמִיעָה אֶת קוֹלָהּ, פָּרְצָה מִפִּיהָ מַקְהֵלָה,
הִיא קִרְקְרָה, הִיא טִרְטְרָה צָוְחָה נָבְחָה וְיִלְּלָה.
נוֹטֶפֶת בֹּץ וּמַיִם, קָרְאָה מִן הַשְּׁלוּלִית:
"עוּף מִפֹּה דְּרָקוֹן, הַמְּכַשֵּׁפָה שֶׁלִּי!"

הַדְּרָקוֹן הַמְּבֹהָל נָסוֹג בִּמְהִירוּת:
"סְלִיחָה..." גִּמְגֵּם הוּא וְרָעַד, "זוֹ כַּנִּרְאֶה טָעוּת,
הָיָה נָעִים מְאוֹד אֲבָל יֵשׁ לִי סִדּוּרִים!"
פָּרַשׂ כָּנָף וְעָף לוֹ, עָף מֵעֵבֶר לֶהָרִים.

וְאָז נִפְנֵף לוֹ הָעוֹרֵב,
נִטְּרָה לָהּ הַצְּפַרְדֵּעַ,
הַכֶּלֶב עוֹד פָּלַט קְצָת בֹּץ,
וְהַחָתוּל קִרְטֵעַ.

"תּוֹדָה!" זִמְּרָה הַמְּכַשֵּׁפָה, "מַה שֶּׁנָּכוֹן נָכוֹן,
בִּלְעֲדֵיכֶם, הָיִיתִי כְּבָר בְּבֶטֶן הַדְּרָקוֹן!"
אָז מִלְּאָה אֶת הַקַּלַּחַת וְהִבְעִירָה אֵשׁ מִתַּחַת,
וּלְחַשׁ קוֹל פִּזְּמָה: "כָּל אֶחָד יִמְצָא דְּבַר מָה!"

זוֹ קָטְפָה לָהּ פֶּרַח בָּר,
זֶה מָצָא לוֹ אִצְטְרֻבָּל,
זֶה עָנָף יָבֵשׁ,
זֶה עֶצֶם,
הִנֵּה כְּבָר מַתְחִיל הַקֶּסֶם!

אֶת הַכֹּל הִשְׁלִיכוּ יַחַד,
חִישׁ מַהֵר אֶל הַקַּלַּחַת,
וּבַשַּׁרְבִיט, הַמְּכַשֵּׁפָה, אֶת כָּל אֵלֶּה עִרְבְּבָה,
קְצָת הוֹסִיפָה, קְצָת טָבְלָה,
לָחֲשָׁה וּמִלְמְלָה:
"הִיגָאטִי, טִיגָאטִי, טָאגָטִי טָה!"
וְהוֹפּ, קָפַץ לוֹ מַטְאֲטֵא!

וְאֵיזֶה מַטְאֲטֵא!
עִם מוֹשָׁב לְכָל נוֹסֵעַ, קֵן לְמַעְלָה לָעוֹרֵב, וּמוֹשָׁב לְכָל נוֹסֵעַ.
"יֵשׁ!" קָרְאָה הַמְּכַשֵּׁפָה חוֹגֶגֶת וְצוֹהֶלֶת,
וְעוּף... כֻּלָּם הִמְרִיאוּ עַל מַטְאֲטֵא הַפֶּלֶא!
